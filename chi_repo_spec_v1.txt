[fn chi_encode_frame]
role = "Encode message type + payload into TLV wire format"
inputs = ["msg_type: MessageType (u8)", "payload: &[u8]"]
outputs = ["Vec<u8>: [type:1][length:4][payload:N]"]
hot_path = true
optimization_notes = [
  "Currently allocates Vec per call",
  "Could write directly to caller-provided buffer",
  "Length is big-endian (network byte order)",
]

[fn chi_decode_frame]
role = "Decode TLV frame from reader"
inputs = ["reader: impl Read"]
outputs = ["(MessageType, Vec<u8>)"]
hot_path = true
optimization_notes = [
  "Currently 3 read_exact() calls (type, length, payload)",
  "Could coalesce to 2 (header + payload)",
  "Allocates Vec for payload each call",
]

[fn chi_write_frame]
role = "Write TLV frame to writer and flush"
inputs = ["writer: impl Write", "msg_type: MessageType", "payload: &[u8]"]
outputs = ["io::Result<()>"]
hot_path = true
optimization_notes = [
  "Calls encode_frame() then write_all() - double handling",
  "Flushes after every frame - expensive on TCP",
  "Could write header+payload directly, let caller flush",
]

# Usage Notes for LLMs

When starting a Chi session:
1. Load this spec (~2K tokens)
2. Select profile matching task (protocol, architecture, rendering, etc.)
3. Load files from that profile's include list
4. Check [phases] for current status and blockers

When modifying Chi:
- Keep flat structure (no src/ nesting)
- Use chi_{category}_{name}.{ext} naming
- Update this spec when adding/removing files
- Add [fn ...] entries for significant functions

Key insight: Chi is the front-end of a semantic OS. Everything visible is a bubble. Spirit doesn't render pixels; it composes meaning.
